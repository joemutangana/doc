Between and in
----------------

select first_name, salary from employees where salary in(5000,80000); return only those values where they are true
select first_name, salary from employees where salary between 5000 and 80000; returns the values fall in that range of numbers

DATE COMPARISON
===============



select first_name, last_name, gpa, status, ENROLLMENT_DATE from students WHERE enrollment_date = DATE'2023-09-01';

EXTRACT YEAR
--------



select first_name, last_name, gpa, status, ENROLLMENT_DATE from students 
WHERE EXTRACT(YEAR FROM ENROLLMENT_DATE) = 2023;

OR USE BETWEEN



select first_name, last_name, gpa, status, ENROLLMENT_DATE from students WHERE enrollment_date BETWEEN DATE'2023-01-01' AND DATE'2023-12-31';
