------------------------------
generate the self signed certificate
----------------------
sudo openssl req -x509 -nodes -days 365 \
  -newkey rsa:2048 \
  -keyout /etc/ssl/private/29061.f26.key \
  -out /etc/ssl/certs/29061.f26.crt \
  -subj "/O=AUCA/OU=Linux-F26/CN=29061" \
  -addext "subjectAltName=DNS:29061.f26"


------------------------------
Verify the certficate
-----------------------------
sudo openssl x509 -in /etc/ssl/certs/29061.f26.crt -noout -subject -issuer -dates

Check the cert domain
------------------------
sudo openssl x509 -in /etc/ssl/certs/29061.f26.crt -noout -ext subjectAltName

virtual host configuration
-----------------------

# HTTP website
<VirtualHost *:80>
    ServerName 29061.f26

    DocumentRoot /var/www/29061

    <Directory /var/www/29061>
        Options -Indexes
        AllowOverride None
        Require all granted
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/29061-http-error.log
    CustomLog ${APACHE_LOG_DIR}/29061-http-access.log combined
</VirtualHost>


# HTTPS website
<VirtualHost *:443>
    ServerName 29061.f26

    DocumentRoot /var/www/29061

    SSLEngine on
    SSLCertificateFile /etc/ssl/29061/29061.crt
    SSLCertificateKeyFile /etc/ssl/29061/29061.key

    <Directory /var/www/29061>
        Options -Indexes
        AllowOverride None
        Require all granted
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/29061-https-error.log
    CustomLog ${APACHE_LOG_DIR}/29061-https-access.log combined
</VirtualHost>

------
Bash to send email on each new file
-------------------------------

#!/bin/bash

INCOMING="$HOME/incoming"
OWNER="joegithub12@gmail.com"
received_at="$(date +'%Y-%m-%d %H:%M:%S')"
source="VM1"

inotifywait --monitor --event close_write --event moved_to --format %f "$INCOMING" |
 while read -r filename; do
   echo "New file detected:\n $filename"
   printf 'New backup received. Filename: %s\n Source: %s\n Received at: %s\n' "$filename" "$source" "$received_at |
mail -s "New backup received $filename" "OWNER"
 done
